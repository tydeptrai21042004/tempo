import os, subprocess
from pathlib import Path
repo=Path('/kaggle/working/tempo')
if not repo.exists():
    subprocess.run(['git','clone','--depth','1','https://github.com/tydeptrai21042004/tempo.git',str(repo)],check=True)
else:
    subprocess.run(['git','fetch','origin','main'],cwd=repo,check=True)
    subprocess.run(['git','reset','--hard','origin/main'],cwd=repo,check=True)
env=os.environ.copy()
env.update(CUDA_VISIBLE_DEVICES='-1',NUMBA_NUM_THREADS='4',OMP_NUM_THREADS='1',
           OPENBLAS_NUM_THREADS='1',MKL_NUM_THREADS='1',PYTHONHASHSEED='0',
           KAGGLE_REPORT_ROOT='/kaggle/working/TEMPO_FULL_RESULTS',
           KAGGLE_SELECTED_DATA_DIR='/kaggle/working/TEMPO_SELECTED_DATA',
           KAGGLE_MASTER_ZIP='/kaggle/working/TEMPO_PIPELINE_RESULTS.zip',
           VIDEO_FRAMES='180',STREAM_FRAMES='180',FAIL_ON_INCOMPLETE='0')
subprocess.run(['bash','scripts/kaggle_auto_video.sh','full'],cwd=repo,env=env,check=True)
print('/kaggle/working/TEMPO_PIPELINE_RESULTS.zip')
